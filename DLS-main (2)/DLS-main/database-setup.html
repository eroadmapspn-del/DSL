Step 2: Database Schema (if tables are missing)

If the status check shows missing tables, copy this SQL and run it in your Supabase SQL Editor:

-- Purchase Tables Schema for Supabase
-- Copy and paste this entire block into your Supabase SQL Editor

-- 1. Purchase Retailers Table
CREATE TABLE IF NOT EXISTS purchase_retailers (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    retailer_name VARCHAR(255) NOT NULL,
    due_days INTEGER NOT NULL DEFAULT 0,
    user_id VARCHAR(255) NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    UNIQUE(retailer_name, user_id)
);

-- 2. Purchase Entries Table  
CREATE TABLE IF NOT EXISTS purchase_entries (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    unique_id VARCHAR(255) NOT NULL,
    bill_number VARCHAR(255) NOT NULL,
    bill_date DATE NOT NULL,
    retailer_name VARCHAR(255) NOT NULL,
    taxable_amount DECIMAL(10,2) NOT NULL DEFAULT 0,
    total_tax DECIMAL(10,2) NOT NULL DEFAULT 0,
    deduction DECIMAL(10,2) NOT NULL DEFAULT 0,
    net_amount DECIMAL(10,2) NOT NULL DEFAULT 0,
    deduction_type VARCHAR(100),
    credit_note_number VARCHAR(255),
    credit_note_date DATE,
    status VARCHAR(50) NOT NULL DEFAULT 'Received',
    note TEXT,
    voucher_number VARCHAR(255),
    confirmed_at TIMESTAMP WITH TIME ZONE,
    completed_at TIMESTAMP WITH TIME ZONE,
    due_date DATE,
    user_id VARCHAR(255) NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    UNIQUE(unique_id, user_id),
    UNIQUE(bill_number, user_id)
);

-- Create indexes for better performance
CREATE INDEX IF NOT EXISTS idx_purchase_retailers_user_id ON purchase_retailers(user_id);
CREATE INDEX IF NOT EXISTS idx_purchase_retailers_retailer_name ON purchase_retailers(retailer_name);
CREATE INDEX IF NOT EXISTS idx_purchase_entries_user_id ON purchase_entries(user_id);
CREATE INDEX IF NOT EXISTS idx_purchase_entries_status ON purchase_entries(status);
CREATE INDEX IF NOT EXISTS idx_purchase_entries_retailer_name ON purchase_entries(retailer_name);
CREATE INDEX IF NOT EXISTS idx_purchase_entries_bill_date ON purchase_entries(bill_date);

-- Enable Row Level Security
ALTER TABLE purchase_retailers ENABLE ROW LEVEL SECURITY;
ALTER TABLE purchase_entries ENABLE ROW LEVEL SECURITY;

-- Create RLS policies for anonymous access (since we're not using auth yet)
DROP POLICY IF EXISTS "Allow anonymous access to purchase_retailers" ON purchase_retailers;
CREATE POLICY "Allow anonymous access to purchase_retailers" ON purchase_retailers
FOR ALL TO anon, authenticated 
USING (true) 
WITH CHECK (true);

DROP POLICY IF EXISTS "Allow anonymous access to purchase_entries" ON purchase_entries;
CREATE POLICY "Allow anonymous access to purchase_entries" ON purchase_entries
FOR ALL TO anon, authenticated  
USING (true) 
WITH CHECK (true);

-- Create trigger for updated_at
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ language 'plpgsql';

DROP TRIGGER IF EXISTS update_purchase_retailers_updated_at ON purchase_retailers;
CREATE TRIGGER update_purchase_retailers_updated_at
    BEFORE UPDATE ON purchase_retailers
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

DROP TRIGGER IF EXISTS update_purchase_entries_updated_at ON purchase_entries;
CREATE TRIGGER update_purchase_entries_updated_at
    BEFORE UPDATE ON purchase_entries
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
How to run this SQL:
1. Go to your Supabase Dashboard
2. Select your project
3. Click "SQL Editor" in the sidebar
4. Click "New Query"
5. Paste the SQL above
6. Click "Run" or press Ctrl+Enter
